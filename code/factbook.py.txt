def num(s):
    if s is None: return None
    s=re.sub(r'\(.*?\)','',str(s)).replace(',','')
    m=re.search(r'(-?\$?\d+(?:\.\d+)?)\s*(trillion|billion|million)?',s)
    if not m: return None
    v=float(m.group(1).replace('$',''))
    return v*{'trillion':1e12,'billion':1e9,'million':1e6,None:1}[m.group(2)]
def get(d,*path):
    for p in path:
        if not isinstance(d,dict) or p not in d: return None
        d=d[p]
    return d.get('text') if isinstance(d,dict) else d
