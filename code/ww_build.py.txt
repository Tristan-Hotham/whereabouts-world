def km(a,b,c,d):
    R=6371;p1,p2=math.radians(a),math.radians(c);dp=p2-p1;dl=math.radians(d-b)
    h=math.sin(dp/2)**2+math.cos(p1)*math.cos(p2)*math.sin(dl/2)**2;return 2*R*math.asin(math.sqrt(h))

def bearing(a,b,c,d):
    y=math.sin(math.radians(d-b))*math.cos(math.radians(c));x=math.cos(math.radians(a))*math.sin(math.radians(c))-math.sin(math.radians(a))*math.cos(math.radians(c))*math.cos(math.radians(d-b))
    return (math.degrees(math.atan2(y,x))+360)%360

def quirk(c,seed=0):
    def find(ids,lim,both):
        best=None
        for k in ids:
            m=BYID[k]
            if c not in m['v'] or len(m['v'])<25: continue
            n=len(m['v']); r=RANK[k][c]
            cands=[(r/n,'hi',r)]+([((n-r+1)/n,'lo',n-r+1)] if both else [])
            for f,s,rr in cands:
                if rr<=max(1,n*lim) and (best is None or f<best[0]): best=(f,k,rr,n,m['v'][c],s)
        return best
    return find(FUNIDS,.05,False) or find(SERIDS,.03,True) or find(FUNIDS,.1,False) or find(SERIDS,.06,True) or find(FUNIDS,.2,False) or find(SERIDS,.12,True)
