def df(page,i):

def fnum(x):
    if x is None or (isinstance(x,float) and x!=x): return None
    s=str(x).replace(',','').replace('\xa0',' ').replace('−','-')
    m=re.search(r'-?\d+(?:\.\d+)?',s); return float(m.group()) if m else None
UNM={}
def series(page,i,namecol,valcol,f=fnum,agg='first'):
    d=df(page,i); out={}
    nc=[c for c in d.columns if namecol in c][0]; vc=[c for c in d.columns if valcol in c][0]
    for _,r in d.iterrows():
        n=str(r[nc]); c=iso(n); v=f(r[vc])
        if c=='?': UNM.setdefault(page,[]).append(n); continue
        if c is None or v is None: continue
        if agg=='sum': out[c]=out.get(c,0)+v
        elif agg=='max': out[c]=max(out.get(c,v),v)
        elif c not in out: out[c]=v
    return out
